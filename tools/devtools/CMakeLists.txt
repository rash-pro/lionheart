cmake_minimum_required(VERSION 3.21)
project(lionheart_devtools LANGUAGES CXX)
add_library(pad-support INTERFACE)
target_include_directories(pad-support INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(pad-support INTERFACE cxx_std_17)
add_executable(iro-extract iro_extract.cpp)
target_compile_features(iro-extract PRIVATE cxx_std_17)
get_filename_component(LH_TOOLS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
target_compile_definitions(iro-extract PRIVATE LH_TOOLS_ROOT="${LH_TOOLS_ROOT}")
add_executable(perf-report perf_report.cpp)
target_compile_features(perf-report PRIVATE cxx_std_17)
add_executable(compare compare.cpp)
target_compile_features(compare PRIVATE cxx_std_17)
add_executable(frame-probe frame_probe.cpp)
target_compile_features(frame-probe PRIVATE cxx_std_17)
target_compile_definitions(frame-probe PRIVATE LH_TOOLS_ROOT="${LH_TOOLS_ROOT}")
add_executable(render-quality render_quality.cpp)
target_compile_features(render-quality PRIVATE cxx_std_17)
add_executable(frame-bench frame_bench.cpp)
target_compile_features(frame-bench PRIVATE cxx_std_17)
add_executable(gte-trace gte_trace.cpp)
target_compile_features(gte-trace PRIVATE cxx_std_17)
target_compile_definitions(gte-trace PRIVATE LH_TOOLS_ROOT="${LH_TOOLS_ROOT}")
add_executable(mdec-trace mdec_trace.cpp)
target_compile_features(mdec-trace PRIVATE cxx_std_17)
target_compile_definitions(mdec-trace PRIVATE LH_TOOLS_ROOT="${LH_TOOLS_ROOT}")
enable_language(C)
add_library(metadata-io STATIC metadata_io.c
  ../../generator/src/base.c ../../generator/src/platform.c
  ../../generator/src/disc.c ../../generator/src/lzs.c ../../generator/src/sha1.c)
target_include_directories(metadata-io PRIVATE ../../generator/src ../../generator/include)
add_executable(ff8files ff8files.cpp)
target_compile_features(ff8files PRIVATE cxx_std_17)
target_compile_definitions(ff8files PRIVATE LH_TOOLS_ROOT="${LH_TOOLS_ROOT}")
target_link_libraries(ff8files PRIVATE metadata-io)
