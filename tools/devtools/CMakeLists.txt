cmake_minimum_required(VERSION 3.21)
project(lionheart_devtools LANGUAGES CXX)
add_library(pad-support INTERFACE)
target_include_directories(pad-support INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(pad-support INTERFACE cxx_std_17)
add_executable(iro-extract iro_extract.cpp)
target_compile_features(iro-extract PRIVATE cxx_std_17)
get_filename_component(LH_TOOLS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
target_compile_definitions(iro-extract PRIVATE LH_TOOLS_ROOT="${LH_TOOLS_ROOT}")
